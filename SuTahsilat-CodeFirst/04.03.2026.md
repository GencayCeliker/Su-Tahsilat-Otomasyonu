   ### Context -Tabloların Kısıtlamalarını yapıyoruz 

```csharp
using Microsoft.EntityFrameworkCore;
using System;
using System.Collections.Generic;
using System.Text;
using SuTahsilat.DAL.Entity;

namespace SuTahsilat.DAL.Context
{
    public class SuTahsilatDBContext :DbContext
    {
        public SuTahsilatDBContext(DbContextOptions<SuTahsilatDBContext> options) : base(options)
        {

        }

        public DbSet<Abone>Aboneler => Set<Abone>();
        public DbSet<Sayac>Sayaclar => Set<Sayac>();
        public DbSet<Tarife>Tarifeler => Set<Tarife>();
        public DbSet<Tuketim>Tuketimler => Set<Tuketim>();
        public DbSet<Ucret>Ucretler => Set<Ucret>();

        protected override void OnModelCreating(ModelBuilder modelBuilder) // modelin oluşturulma esnasında nasıl davranacağı
        {
            //ABONE Tablosu
            modelBuilder.Entity<Abone>(e=>
            {
                e.ToTable("ABONE");//Tablo Adı
                e.HasKey(x=>x.AboneID);// Primary Key

                //ÖZELLİK              NOT NULL     VARCHAR(200)     NVARCHAR
                e.Property(x => x.Ad).IsRequired().HasMaxLength(200).IsUnicode();

                e.Property(x => x.Soyad).IsRequired().HasMaxLength(200).IsUnicode();
                e.Property(x=>x.TCKimlikNo).IsRequired().HasMaxLength(11);
                e.HasIndex(x => x.TCKimlikNo).IsUnique();
                e.Property(x=>x.Adres).IsRequired().HasMaxLength(500).IsUnicode();
                e.Property(x => x.Telefon).IsRequired().HasMaxLength(15);
                
            });
            //********************************************************/

            //Sayac Tablosu
            modelBuilder.Entity<Sayac>(e =>
            {
                e.ToTable("SAYAC");//Tablo Adı
                e.HasKey(x => x.SayacID);// Primary Key

                //ÖZELLİK              NOT NULL         VARCHAR(50)     NVARCHAR
                e.Property(x => x.Marka).IsRequired().HasMaxLength(50).IsUnicode();
                e.Property(x => x.SeriNo).IsRequired().HasMaxLength(50);
                e.HasIndex(x => x.SeriNo).IsUnique();
                //Bir abone nin birden fazla sayacı olabilir
                e.HasOne(x => x.Abone).WithMany(a=>a.Sayaclar).HasForeignKey(x=>x.AboneID).OnDelete(DeleteBehavior.Restrict);
                e.HasOne(x => x.Tarife).WithMany(a => a.Sayaclar).HasForeignKey(x => x.TarifeID).OnDelete(DeleteBehavior.Restrict);

                

            });
            //********************************************************/

            //Tuketim Tablosu
            modelBuilder.Entity<Tuketim>(e =>
            {
                e.ToTable("TUKETIM");//Tablo Adı
                e.HasKey(x => x.TuketimID);// Primary Key

                e.Property(x => x.TuketimTarihi).IsRequired();
                e.Property(x => x.Endeks).IsRequired();
                e.Property(x => x.Tutar).IsRequired().HasColumnType("decimal(20,2)");

                e.HasOne(x => x.Sayac).WithMany(a => a.Tuketimler).HasForeignKey(x => x.SayacID).OnDelete(DeleteBehavior.Restrict);

            });
            //********************************************************/

            //Ucret Tablosu
            modelBuilder.Entity<Ucret>(e =>
            {
                e.ToTable("UCRET");//Tablo Adı
                e.HasKey(x => x.UcretID);// Primary Key

                e.Property(x => x.GecerliTarih).IsRequired();
                e.Property(x => x.Birimfiyat).IsRequired().HasColumnType("decimal(20,2)");
                e.Property(x => x.CTVOrani).IsRequired().HasColumnType("decimal(3,2)");
                e.Property(x => x.ABOrani).IsRequired().HasColumnType("decimal(3,2)");
                e.Property(x => x.KDVOrani).IsRequired().HasColumnType("decimal(3,2)");

                e.HasOne(x => x.Tarife).WithMany(a => a.Ucretler).HasForeignKey(x => x.TarifeID).OnDelete(DeleteBehavior.Restrict);

            });
            //********************************************************/

            //Tarife Tablosu
            modelBuilder.Entity<Tarife>(e =>
            {
                e.ToTable("TARİFE");//Tablo Adı
                e.HasKey(x => x.TarifeID);// Primary Key

                e.Property(x => x.TarifeAdi).IsRequired().HasMaxLength(50).IsUnicode();

            });
            //********************************************************/
            base.OnModelCreating(modelBuilder);
        }
    }
}
```


### Context - SuTahsilatDbContextFactory Adında Sınıf oluşturuyoruz
```csharp
using System;
using System.Collections.Generic;
using System.Text;
using Microsoft.EntityFrameworkCore;
using Microsoft.EntityFrameworkCore.Design;

namespace SuTahsilat.DAL.Context
{
    public class SuTahsilatDbContextFactory : IDesignTimeDbContextFactory<SuTahsilatDBContext>
    {
        public SuTahsilatDBContext CreateDbContext(string[] agrs) 
        {
            var optionbuilder=new DbContextOptionsBuilder<SuTahsilatDBContext>();
            var connStr = "Server=.;" +
                          "Database=SUTAHSILAT_BGRUBU;" +
                          "Trusted_Connection=true;" +
                          "TrustServerCertificate=true";
            optionbuilder.UseSqlServer(connStr);    
            return new SuTahsilatDBContext(optionbuilder.Options);
        }
    }
}
```
